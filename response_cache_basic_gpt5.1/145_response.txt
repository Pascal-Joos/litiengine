  public static BufferedImage horizontalFlip(final BufferedImage img) {
    final int w = img.getWidth();
    final int h = img.getHeight();
    if (w == 0 || h == 0) {

    }

    final BufferedImage dimg = getCompatibleImage(w, h);
    if (dimg == null) {
      return img;
    }
